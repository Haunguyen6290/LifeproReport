// src/lib/finance.ts
export type TkNhom = 'Doanh thu' | 'Trả lại' | 'Thu tiền';
export type TkMapItem = { ma: string; nhom: TkNhom };

/** Excel serial → ms */
const EXCEL_EPOCH = 25569; // ngày 1970-01-01 trong hệ 1900
const DAY_MS = 86400 * 1000;

export function parseNgay(v: unknown): string | null {
  if (v == null || v === '') return null;
  if (typeof v === 'number' && v > 20000) {
    const d = new Date(Math.round((v - EXCEL_EPOCH) * DAY_MS));
    return d.toISOString().slice(0, 10);
  }
  if (v instanceof Date && !isNaN(v.getTime())) return v.toISOString().slice(0, 10);
  const s = String(v).trim();
  if (/^\d{4}-\d{2}-\d{2}/.test(s)) return s.slice(0, 10);
  const m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  if (m) return `${m[3]}-${m[2].padStart(2, '0')}-${m[1].padStart(2, '0')}`;
  const d = new Date(s);
  return isNaN(d.getTime()) ? null : d.toISOString().slice(0, 10);
}

/** Khớp tiền tố dài nhất trong map */
export function phanLoaiTk(tkDoiUng: string, map: TkMapItem[]): TkNhom | null {
  const tk = String(tkDoiUng ?? '').trim();
  let best: { len: number; nhom: TkNhom } | null = null;
  for (const it of map) {
    if (!it.ma) continue;
    if (tk.startsWith(it.ma) && (!best || it.ma.length > best.len)) best = { len: it.ma.length, nhom: it.nhom };
  }
  return best ? best.nhom : null;
}
```

-- ===== 6) RPC báo cáo nợ quá hạn =====
create or replace function public.finance_debt_report(p_thang text, p_han int default 90)
returns json language plpgsql stable as $$
declare
  v_base date; v_D date; v_E date; v_map jsonb;
  v_rows json;
begin
  select value::date into v_base from public.settings where key='DEBT_BASE_DATE';
  if v_base is null then v_base := '2026-01-01'; end if;
  -- D = đầu tháng của (cuối tháng xem − hạn)
  v_D := date_trunc('month', (date_trunc('month', (p_thang || '-01')::date) + interval '1 month' - interval '1 day') - (p_han || ' days')::interval)::date;
  -- E = ngày cuối có chứng từ (fallback cuối tháng xem)
  select coalesce(max(ngay), (date_trunc('month',(p_thang||'-01')::date)+interval '1 month'-interval '1 day')::date) into v_E from public.receivable_rows;
  select coalesce(value::jsonb,'[]'::jsonb) into v_map from public.settings where key='RECEIVABLE_TK_MAP';

  with base as (
    select cb.ma_kh, cb.ten_kh, cb.du_no,
           coalesce(c.assigned, '') as nvkd,
           coalesce(c.tinh_thanh, '') as tinh
    from public.customer_base_balance cb
    left join (
      select c2.ma_kh, p.full_name as assigned, c2.tinh_thanh
      from public.customers c2 left join public.profiles p on p.id = c2.assigned_to
    ) c on c.ma_kh = cb.ma_kh
  ),
  ps as (
    select ma_kh,
      sum(case when ngay < v_D then so_no - so_co else 0 end) as ps_truoc_D,
      sum(case when ngay >= v_D and ngay <= v_E and public.fn_tk_nhom(tk_doi_ung, v_map)='Doanh thu' then so_no - so_co else 0 end) as doanh_thu,
      sum(case when ngay >= v_D and ngay <= v_E and public.fn_tk_nhom(tk_doi_ung, v_map)='Trả lại' then so_co - so_no else 0 end) as tra_lai,
      sum(case when ngay >= v_D and ngay <= v_E and public.fn_tk_nhom(tk_doi_ung, v_map)='Thu tiền' then so_co - so_no else 0 end) as thu_tien
    from public.receivable_rows
    group by ma_kh
  )
  select json_agg(t order by (t->>'con_thieu')::numeric desc, (t->>'ma_kh')::text) into v_rows
  from (
    select json_build_object(
      'ma_kh', b.ma_kh, 'ten_kh', b.ten_kh, 'nvkd', b.nvkd, 'tinh', b.tinh,
      'cong_no_dau_ky', greatest(b.du_no + coalesce(p.ps_truoc_D,0), 0),
      'doanh_thu', coalesce(p.doanh_thu,0),
      'tra_lai', coalesce(p.tra_lai,0),
      'thu_tien', coalesce(p.thu_tien,0),
      'tong_giam_tru', coalesce(p.tra_lai,0) + coalesce(p.thu_tien,0),
      'con_thieu', greatest(b.du_no + coalesce(p.ps_truoc_D,0) - (coalesce(p.tra_lai,0)+coalesce(p.thu_tien,0)), 0),
      'qua_han', (coalesce(p.tra_lai,0)+coalesce(p.thu_tien,0)) < greatest(b.du_no + coalesce(p.ps_truoc_D,0), 0) and (b.du_no + coalesce(p.ps_truoc_D,0)) > 0
    ) as t
    from base b left join ps p on p.ma_kh = b.ma_kh
  ) t;

  return json_build_object('D', v_D, 'E', v_E, 'base', v_base, 'rows', coalesce(v_rows,'[]'::json));
end;
$$;

-- helper phân loại TK trong SQL (tiền tố dài nhất)
create or replace function public.fn_tk_nhom(tk text, map jsonb) returns text language sql immutable as $$
  select elem->>'nhom' from jsonb_array_elements(map) elem
  where tk like (elem->>'ma') || '%'
  order by length(elem->>'ma') desc limit 1
$$;

-- ===== 7) RPC báo cáo thu tiền theo tháng =====
create or replace function public.finance_collections_report(p_thang text)
returns json language plpgsql stable as $$
declare
  v_tu date; v_den date; v_map jsonb; v_plan jsonb;
  v_rows json;
begin
  v_tu := (p_thang || '-01')::date;
  v_den := (date_trunc('month', v_tu) + interval '1 month' - interval '1 day')::date;
  select coalesce(value::jsonb,'[]'::jsonb) into v_map from public.settings where key='RECEIVABLE_TK_MAP';
  select coalesce(value::jsonb,'[]'::jsonb) into v_plan from public.settings where key='FINANCE_PLAN';

  with kh_nv as (  -- quy NVKD theo hồ sơ khách hàng
    select s.ma_kh, s.thanh_tien, s.ngay,
           coalesce(p.full_name, nullif(s.kinh_doanh,'')) as nvkd
    from public.sales_rows s
    left join public.customers c on c.ma_kh = s.ma_kh
    left join public.profiles p on p.id = c.assigned_to
    where s.ngay between v_tu and v_den
  ),
  thu as (  -- thu tiền từ TK131 nhóm Thu tiền
    select r.ma_kh, sum(r.so_co - r.so_no) as so_thu
    from public.receivable_rows r
    where r.ngay between v_tu and v_den
      and public.fn_tk_nhom(r.tk_doi_ung, v_map) = 'Thu tiền'
    group by r.ma_kh
  ),
  khuvu as (
    select coalesce(k.nvkd, 'Khác') as nvkd,
           sum(k.thanh_tien) as doanh_so,
           coalesce(sum(t.so_thu),0) as thu_tien
    from kh_nv k left join thu t on t.ma_kh = k.ma_kh
    group by 1
  )
  select json_agg(t) into v_rows from (
    select json_build_object('nvkd', nvkd, 'doanh_so', doanh_so, 'thu_tien', thu_tien) as t from khuvu
  ) t;

  return json_build_object('thang', p_thang, 'plan', v_plan, 'rows', coalesce(v_rows,'[]'::json));
end;
$$;
```

// src/lib/receivable-import.ts
import { parseNgay } from '@/lib/finance';

export type RcvRow = {
  ngay: string; so_ct: string; ma_kh: string; ten_kh: string;
  dien_giai: string; tk_doi_ung: string; so_no: number; so_co: number; du_dong: number | null;
};

const num = (v: unknown): number => {
  if (typeof v === 'number') return v;
  if (v == null || v === '') return 0;
  const n = Number(String(v).replace(/[,\s]/g, ''));
  return isNaN(n) ? 0 : n;
};

function findSheetDataStart(rows: unknown[][]): number {
  // dòng có "Số hiệu" + "Ngày" là dòng tiêu đề thứ 2 → dữ liệu từ dòng kế tiếp
  for (let i = 0; i < Math.min(rows.length, 30); i++) {
    const line = rows[i].map((c) => String(c ?? '').toLowerCase());
    if (line.some((c) => c.includes('số hiệu')) && line.some((c) => c.includes('ngày'))) return i + 1;
  }
  return 0;
}

export function parseTk131Sheet(rows: unknown[][]): { header: { tuNgay: string; denNgay: string; soDuDauKy: number }; rows: RcvRow[] } {
  let tuNgay = '', denNgay = '', soDuDauKy = 0;
  for (const r of rows) {
    const txt = r.map((c) => String(c ?? '')).join(' ');
    const m = txt.match(/Từ ngày[:\s]*([\d\/\-]+)\s*đến ngày[:\s]*([\d\/\-]+)/i);
    if (m && !tuNgay) { tuNgay = parseNgay(m[1]) ?? ''; denNgay = parseNgay(m[2]) ?? ''; }
    if (/Số dư đầu kỳ/i.test(txt)) {
      for (let i = 7; i < r.length; i++) { const v = num(r[i]); if (v !== 0) { soDuDauKy = v; break; } }
    }
  }
  const start = findSheetDataStart(rows);
  const out: RcvRow[] = [];
  for (let i = start; i < rows.length; i++) {
    const r = rows[i];
    const tk = String(r[0] ?? '').trim();
    const ngay = parseNgay(r[2]);
    if (tk !== '131' || !ngay) continue;
    out.push({
      ngay,
      so_ct: String(r[1] ?? '').trim(),
      ma_kh: String(r[3] ?? '').trim(),
      ten_kh: String(r[4] ?? '').trim(),
      dien_giai: String(r[5] ?? '').trim(),
      tk_doi_ung: String(r[6] ?? '').trim(),
      so_no: num(r[7]),
      so_co: num(r[8]),
      du_dong: r[9] != null ? num(r[9]) : null,
    });
  }
  return { header: { tuNgay, denNgay, soDuDauKy }, rows: out };
}

export function parseDataKHSheet(rows: unknown[][]): { ma: string; ten: string; duNo: number }[] {
  // header dòng 0; cột 6 (hoặc cột cuối) là dư nợ đầu kỳ
  if (!rows.length) return [];
  let colDu = -1;
  const h = rows[0].map((c) => String(c ?? '').toLowerCase());
  h.forEach((c, i) => { if (c.includes('nợ') || c.includes('du no')) colDu = i; });
  if (colDu === -1) colDu = h.length - 1;
  const out: { ma: string; ten: string; duNo: number }[] = [];
  for (let i = 1; i < rows.length; i++) {
    const ma = String(rows[i][0] ?? '').trim();
    if (!ma) continue;
    out.push({ ma, ten: String(rows[i][1] ?? '').trim(), duNo: num(rows[i][colDu]) });
  }
  return out;
}

function findSheet(wb: XLSX.WorkBook): string | null {
  const cand = wb.SheetNames.find((n) => /131|tk\s*131|tai_khoan/i.test(n) && !/helper/i.test(n));

export async function POST(req: NextRequest) {
  try {
    const form = await req.formData();
    const file = form.get('file') as File | null;
    const mode = String(form.get('mode') ?? 'preview');
    if (!file) return NextResponse.json({ error: 'Chưa có file' }, { status: 400 });

    const wb = XLSX.read(Buffer.from(await file.arrayBuffer()), { type: 'buffer' });
    const sheetName = findSheet(wb);
    if (!sheetName) return NextResponse.json({ error: 'File không có sheet' }, { status: 400 });
    const raw = XLSX.utils.sheet_to_json(wb.Sheets[sheetName], { header: 1, defval: null }) as unknown[][];
    const { header, rows } = parseTk131Sheet(raw);
    if (rows.length === 0) return NextResponse.json({ error: 'Không đọc được dòng phát sinh nào từ sheet TK131' }, { status: 400 });

    const kiemTra: { ten: string; ok: boolean; chiTiet: string }[] = [];
    const canhBao: string[] = [];

    // K1: cân đối nội bộ vùng mới = số dư đầu file + ΣNợ − ΣCó phải bằng số dư dòng cuối
    const sumNo = rows.reduce((a, r) => a + r.so_no, 0);
    const sumCo = rows.reduce((a, r) => a + r.so_co, 0);
    const duCuoiFile = rows[rows.length - 1].du_dong;
    const tinhRa = header.soDuDauKy + sumNo - sumCo;
    const k1ok = duCuoiFile == null || Math.abs(tinhRa - duCuoiFile) < 1;
    kiemTra.push({ ten: 'Cân đối nội bộ sổ', ok: k1ok, chiTiet: `Đầu kỳ ${header.soDuDauKy} + Nợ ${sumNo} − Có ${sumCo} = ${tinhRa}` + (duCuoiFile != null ? ` (cuối sổ ${duCuoiFile})` : '') });

    // K3: tổng đầu kỳ theo khách (nếu có DataKH) vs toàn công ty
    const dkhSheet = wb.SheetNames.find((n) => /datakh|data kh/i.test(n));
    let baseRows: { ma: string; ten: string; duNo: number }[] = [];
    if (dkhSheet) {
      baseRows = parseDataKHSheet(XLSX.utils.sheet_to_json(wb.Sheets[dkhSheet], { header: 1, defval: null }) as unknown[][]);
      const sumBase = baseRows.reduce((a, b) => a + b.duNo, 0);
      kiemTra.push({ ten: 'Tổng đầu kỳ theo khách = đầu kỳ sổ', ok: Math.abs(sumBase - header.soDuDauKy) < 1, chiTiet: `Σ khách ${sumBase} so với sổ ${header.soDuDauKy}` });
    }

    if (!kiemTra.every((k) => k.ok)) {
      return NextResponse.json({ preview: true, blocked: true, soDong: rows.length, tuNgay: header.tuNgay, denNgay: header.denNgay, kiemTra, canhBao }, { status: 200 });
    }

    if (mode !== 'commit') {
      return NextResponse.json({ preview: true, soDong: rows.length, tuNgay: header.tuNgay, denNgay: header.denNgay, kiemTra, canhBao });
    }

    // ===== COMMIT =====
    const db = admin();
    const batch = new Date().toISOString().slice(0, 19).replace(/[:T]/g, '-');
    const minNgay = header.tuNgay || rows[0].ngay;
    const maxNgay = header.denNgay || rows[rows.length - 1].ngay;

    // K2: liền mạch — số dư đầu file phải bằng hệ thống tính tại minNgay (nếu đã có dữ liệu trước đó)
    const { data: settingBase } = await db.from('settings').select('value').eq('key', 'DEBT_BASE_DATE').maybeSingle();
    const baseDate = (settingBase as any)?.value ?? '2026-01-01';
    const { data: prior } = await db.from('receivable_rows').select('so_no, so_co').lt('ngay', minNgay);
    if ((prior ?? []).length > 0) {
      const { data: baseBal } = await db.from('customer_base_balance').select('du_no');
      const heThong = (baseBal ?? []).reduce((a, r: any) => a + Number(r.du_no || 0), 0)
        + (prior ?? []).reduce((a, r: any) => a + Number(r.so_no) - Number(r.so_co), 0);
      if (Math.abs(heThong - header.soDuDauKy) >= 1) {
        return NextResponse.json({ blocked: true, preview: true, soDong: rows.length, tuNgay: header.tuNgay, denNgay: header.denNgay,
          kiemTra: [...kiemTra, { ten: 'Liền mạch dòng thời gian', ok: false, chiTiet: `Hệ thống tính ${heThong} tại ${minNgay}, file khai ${header.soDuDauKy} — chạy lại file phủ từ thời điểm điều chỉnh` }], canhBao }, { status: 200 });
      }
    }
    kiemTra.push({ ten: 'Liền mạch dòng thời gian', ok: true, chiTiet: prior?.length ? `Khớp tại ${minNgay}` : 'Chưa có dữ liệu trước đó — bỏ qua' });

    // Xóa vùng [minNgay,maxNgay] rồi chèn lại
    await db.from('receivable_rows').delete().gte('ngay', minNgay).lte('ngay', maxNgay);
    const insertRows = rows.map((r) => ({ ...r, import_batch: batch }));
    const CHUNK = 2000;
    for (let i = 0; i < insertRows.length; i += CHUNK) {
      const { error } = await db.from('receivable_rows').insert(insertRows.slice(i, i + CHUNK));
      if (error) return NextResponse.json({ error: error.message }, { status: 500 });
    }

    // Ghi số dư gốc nếu có DataKH
    if (baseRows.length) {
      const upserts = baseRows.map((b) => ({ ma_kh: b.ma, ten_kh: b.ten, du_no: b.duNo, ngay_moc: baseDate, updated_at: new Date().toISOString() }));
      for (let i = 0; i < upserts.length; i += CHUNK) {
        const { error } = await db.from('customer_base_balance').upsert(upserts.slice(i, i + CHUNK), { onConflict: 'ma_kh' });
        if (error) return NextResponse.json({ error: error.message }, { status: 500 });
      }
    }

    // Cảnh báo mềm: mã chưa có trong customers; tên trùng nhiều mã
    const { data: kh } = await db.from('customers').select('ma_kh, ten_kh');
    const maSet = new Set((kh ?? []).map((r: any) => String(r.ma_kh).trim()));
    const laTrongSo = [...new Set(rows.map((r) => r.ma_kh).filter(Boolean))].filter((m) => !maSet.has(m));
    if (laTrongSo.length) canhBao.push(`${laTrongSo.length} mã KH chưa có trong danh mục: ${laTrongSo.slice(0, 10).join(', ')}${laTrongSo.length > 10 ? '…' : ''}`);
    const byName = new Map<string, Set<string>>();
    for (const r of rows) { if (!r.ma_kh) continue; const k = r.ten_kh.toLowerCase(); byName.set(k, (byName.get(k) ?? new Set()).add(r.ma_kh)); }
    const trungTen = [...byName.values()].filter((s) => s.size > 1).length;
    if (trungTen) canhBao.push(`${trungTen} tên KH đang gắn nhiều mã — kiểm tra danh mục`);

    return NextResponse.json({ preview: false, soDong: rows.length, tuNgay: header.tuNgay, denNgay: header.denNgay, kiemTra, canhBao });
  } catch (e: any) {
    return NextResponse.json({ error: e?.message ?? 'Lỗi không xác định' }, { status: 500 });
  }
}
